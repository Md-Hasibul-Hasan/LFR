void JunctionState()
{
    /*
    ==========================================================
                        JUNCTION STATE
    ==========================================================

    This state is entered after FOLLOW_LINE detects a junction.

    Responsibilities
    ----------------
    1. Confirm junction.
    2. Identify junction type.
    3. Select path according to competition rule.
    4. Execute the selected movement.
    5. Detect line again.
    6. Return to FOLLOW_LINE.

    ----------------------------------------------------------
    Step 1 : Confirm Junction
    ----------------------------------------------------------
    Avoid false detection by checking the sensor pattern
    for a short period or several consecutive readings.

    ----------------------------------------------------------
    Step 2 : Detect Junction Type
    ----------------------------------------------------------
    Examples

        +
        Cross Junction

        ├──
        Left Branch

        ──┤
        Right Branch

        ┴
        T Junction

    Future function

        JunctionType DetectJunction();

    ----------------------------------------------------------
    Step 3 : Apply Navigation Rule
    ----------------------------------------------------------
    Decide which direction to take.

    Example

        LEFT_PRIORITY
        RIGHT_PRIORITY
        STRAIGHT_PRIORITY

    Future

        DecideJunctionAction();

    ----------------------------------------------------------
    Step 4 : Execute Movement
    ----------------------------------------------------------
    Possible actions

        Go Straight
        Turn Left
        Turn Right
        U Turn

    Can reuse

        TurnLeftState()
        TurnRightState()

    ----------------------------------------------------------
    Step 5 : Recover Line
    ----------------------------------------------------------
    Continue moving until

        !IsLineLost()

    or

        Center sensors detect line.

    ----------------------------------------------------------
    Step 6 : Return
    ----------------------------------------------------------

        robotState = FOLLOW_LINE;

    ==========================================================
    */

    robotState = RobotState::FOLLOW_LINE;
}

void JunctionState()
{
    bool left   = norValue[0] > 500 && norValue[1] > 500;
    bool center = norValue[3] > 500 && norValue[4] > 500;
    bool right  = norValue[6] > 500 && norValue[7] > 500;

    // Cross Junction (+) -> Go Straight
    if (left && center && right)
    {
        robotState = RobotState::FOLLOW_LINE;     
    }

    // Left Branch
    else if (left && center)
    {
        turnStartTime = millis();
        robotState = RobotState::TURN_LEFT;
    }

    // Right Branch
    else if (center && right)
    {
        turnStartTime = millis();
        robotState = RobotState::TURN_RIGHT;
    }



    // T Junction
    else if (!left && center && !right)
    {
        switch (llrIdx)
        {
        case 0: // Left
            turnStartTime = millis();
            robotState = RobotState::TURN_LEFT;
            break;

        case 1: // Right
            turnStartTime = millis();
            robotState = RobotState::TURN_RIGHT;
            break;

        case 2: // Straight
        default:
            robotState = RobotState::FOLLOW_LINE;
            break;
        }
    }

    // Unknown -> Go Straight
    else
    {
        robotState = RobotState::FOLLOW_LINE;
    }
}
